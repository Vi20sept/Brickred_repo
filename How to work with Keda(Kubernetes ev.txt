How to work with Keda(Kubernetes event driven autoscaling) in K8s 
1)Connect to aks(D4ds_v5)
2)Install Keda
helm repo add kedacore https://kedacore.github.io/charts
helm repo update

install
helm install keda kedacore/keda --namespace keda --create-namespace

verify
kubectl get pods -n keda

Create Storage Account
Create Queue
$STORAGE_KEY = az storage account keys list `
-g kedarg01 `
-n kedademostorage01 `
--query "[0].value" `
-o tsv


az storage queue create `
--name orders `
--account-name kedademostorage01 `
--account-key $STORAGE_KEY


Create Kubernetes Secret
kubectl create secret generic azure-storage-secret -n keda-demo --from-literal=connection="<paste-connection-string>"

kubectl create namespace keda-demo

kubectl create secret generic azure-storage-secret -n keda-demo --from-literal=connection="DefaultEndpointsProtocol=https;AccountName=kedademostorage01;AccountKey=Rd6if7E5hYKdbr3EqKzAWm4Csk9F4uqjWiJ81O3IaBI7cNw4bBI71g7rl05YZecEyWvGVYfbbN26+AStGUJN3A==;EndpointSuffix=core.windows.net"
Verify
kubectl get secret -n keda-demo


Deploy Worker Application

nano worker-deployment.yml

apiVersion: apps/v1
kind: Deployment

metadata:
  name: worker
  namespace: keda-demo

spec:
  replicas: 0

  selector:
    matchLabels:
      app: worker

  template:
    metadata:
      labels:
        app: worker

    spec:
      containers:
      - name: worker

        image: busybox

        command:
        - sh
        - -c

        - |
          while true
          do
            echo "Processing messages..."
            sleep 30
          done


kubectl apply -f worker-deployment.yml

kubectl get pods -n keda-demo


Create TriggerAuthentication

triggerauth.yml

apiVersion: keda.sh/v1alpha1
kind: TriggerAuthentication

metadata:
  name: azure-auth
  namespace: keda-demo

spec:
  secretTargetRef:
  - parameter: connection
    name: azure-storage-secret
    key: connection

kubectl apply -f triggerauth.yml


Create ScaledObject

scaledobject.yml

apiVersion: keda.sh/v1alpha1
kind: ScaledObject

metadata:
  name: queue-scaler
  namespace: keda-demo

spec:
  scaleTargetRef:
    name: worker

  minReplicaCount: 0

  maxReplicaCount: 10

  pollingInterval: 10

  cooldownPeriod: 30

  triggers:
  - type: azure-queue

    metadata:
      queueName: orders
      queueLength: "5"

    authenticationRef:
      name: azure-auth

kubectl apply -f scaledobject.yml


Verify Keda

kubectl get scaledobject -n keda-demo

kubectl get hpa -n keda-demo


Watch Pods
kubectl get pods -n keda-demo -w
 no resources found


Add Messages to Queue

for ($i = 1; $i -le 20; $i++) {
    az storage message put `
      --queue-name orders `
      --content "order$i" `
      --account-name kedademostorage01 `
      --account-key $STORAGE_KEY
}


Observe Scaling
kubectl get deployment worker -n keda-demo

or
kubectl get pods -n keda-demo

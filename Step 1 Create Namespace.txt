Step 1 Create Namespace
kubectl create namespace demo


Step 2 Deploy MySQL Backend
mysql-deployment.yml

apiVersion: apps/v1
kind: Deployment

metadata:
  name: mysql
  namespace: demo

spec:
  replicas: 1

  selector:
    matchLabels:
      app: mysql

  template:
    metadata:
      labels:
        app: mysql

    spec:
      containers:
      - name: mysql

        image: mysql:8.0

        env:
        - name: MYSQL_ROOT_PASSWORD
          value: root123

        - name: MYSQL_DATABASE
          value: appdb

        ports:
        - containerPort: 3306




Now apply
kubectl apply -f mysql-deployment.yml



Step 3 Create MySQL Service

mysql-service.yml

apiVersion: v1
kind: Service

metadata:
  name: mysql-service
  namespace: demo

spec:
  selector:
    app: mysql

  ports:
  - port: 3306
    targetPort: 3306

  type: ClusterIP


now apply

kubectl apply -f mysql-service.yml



Step 4 Create Frontend Flask App

create
app.py

from flask import Flask,request
import mysql.connector

app=Flask(__name__)

@app.route("/")
def home():
    return """
    <form method='post' action='/save'>
    Name:<input name='name'>
    <input type='submit'>
    </form>
    """

@app.route("/save",methods=["POST"])
def save():

    name=request.form["name"]

    conn=mysql.connector.connect(
        host="mysql-service",
        user="root",
        password="root123",
        database="appdb"
    )

    cursor=conn.cursor()

    cursor.execute("""
    CREATE TABLE IF NOT EXISTS users(
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100))
    """)

    cursor.execute(
        "INSERT INTO users(name) VALUES(%s)",
        (name,)
    )

    conn.commit()

    cursor.close()
    conn.close()

    return f"{name} inserted successfully"

if __name__=="__main__":
    app.run(host="0.0.0.0",port=5000)
